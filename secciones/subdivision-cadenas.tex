\section{El operador subdivisión}

% ------------------------------------------------------------------------
\subsection{Naturalidad en el caso lineal}

\begin{frame}[allowframebreaks]
  \frametitle{Las aplicaciones lineales respetan la subdivisión}

  Sean $Y$ e $Y'$ convexos y $f\colon Y\to Y'$ lineal.
  El morfismo de complejos de cadenas inducido $f_\#\colon\widetilde{\Sing}_\bullet(Y)\to\widetilde{\Sing}_\bullet(Y')$ preserva símplices singulares lineales, es más $f_\#([v_0,\ldots,v_n])=[f(v_0),\ldots,f(v_n)]$, así que se restringe a
  \[
    f_\#\colon\widetilde{\LC}_\bullet(Y)\longrightarrow\widetilde{\LC}_\bullet(Y').
  \]
  Este morfismo es compatible con el operador cono en el siguiente sentido
  \[f_\#b=f(b)f_\#,\]
  ya que dado un símplice singular lineal $[w_0,\ldots,w_n]$, se tiene que
  \begin{align*}
    f_\#b[w_0,\ldots,w_n]&=f_\#[b,w_0,\ldots,w_n]
      =[f(b),f(w_0),\ldots,f(w_n)]\\
    &=f(b)[f(w_0),\ldots,f(w_n)]
      =f(b)f_\#[w_0,\ldots,w_n].
  \end{align*}
  Además, por linealidad de $f$, se tiene que $f(b_\sigma)=b_{f_\#\sigma}$ para todo símplice singular
  lineal $\sigma$.

  \framebreak

  \begin{lemma}[de naturalidad lineal]
    En $\widetilde{\LC}_\bullet(Y)$, se tiene que $f_\#\Sd=\Sd f_\#$ y $f_\#T=Tf_\#$.
  \end{lemma}

  \begin{proof}
    Por inducción en $n\geq-1$ sobre un símplice singular lineal $\sigma$ de
    dimensión $n$; en grado $-1$ es trivial pues $S=\id$ y $Y=0$. Si $n\geq0$, usando la hipótesis de inducción y que $f_\#$ conmuta con $\partial$ deducimos que
    \begin{align*}
      f_\#\Sd\sigma&=f_\#b_\sigma\Sd\partial\sigma=f(b_\sigma)\,f_\#\Sd\partial\sigma
        =b_{f_\#\sigma}\,\Sd\partial f_\#\sigma=\Sd f_\#\sigma,\\
      f_\#T\sigma&=f_\#b_\sigma(\sigma-T\partial\sigma)=f(b_\sigma)\,f_\#(\sigma-T\partial\sigma)
        =b_{f_\#\sigma}\bigl(f_\#\sigma-T\partial f_\#\sigma\bigr)=Tf_\#\sigma.
    \end{align*}
  \end{proof}
\end{frame}

\subsection{Definición en cadenas singulares}

\begin{frame}
  \frametitle{El operador subdivisión en cadenas singulares}

  La identidad $\id_{\simplex{n}}=[e_0,\ldots,e_n]$ es un símplice singular
  lineal en el convexo $\simplex{n}\subseteq\R^{n+1}$, luego ya tenemos
  $\Sd\id_{\simplex{n}}\in\LC_n(\simplex{n})$ y
  $T\id_{\simplex{n}}\in\LC_{n+1}(\simplex{n})$.

  \begin{definition}
    Sea $X$ un espacio topológico y $\sigma\colon\simplex{n}\to X$ un
    símplice singular. Definimos
    \[
      \Sd\sigma=\sigma_\#(\Sd\id_{\simplex{n}})\in\Sing_n(X),
      \qquad
      T\sigma=\sigma_\#(T\id_{\simplex{n}})\in\Sing_{n+1}(X),
    \]
    y extendemos por linealidad a
    $\Sd\colon\Sing_n(X)\to\Sing_n(X)$ y $T\colon\Sing_n(X)\to\Sing_{n+1}(X)$.
  \end{definition}

  \medskip

  Es decir, se subdivide el símplice estándar y se lleva el resultado a $X$
  mediante $\sigma$.
\end{frame}

\begin{frame}
  \frametitle{Compatibilidad con el caso lineal}

  \begin{lemma}[de compatibilidad]
    Sea $\sigma\colon\simplex{n}\to X$ un símplice singular y
    $\lambda\colon\simplex{m}\to\simplex{n}$ un símplice singular lineal.
    Entonces
    \[
      \Sd(\sigma\lambda)=\sigma_\#\Sd\lambda,
      \qquad
      T(\sigma\lambda)=\sigma_\#T\lambda.
    \]
    En particular, para $\sigma=\id_{\simplex{n}}$, $\Sd$ y $T$ coinciden
    sobre $\LC_\bullet(\simplex{n})$ con los operadores lineales.
  \end{lemma}

  \begin{proof}
    Como $\lambda$ es la restricción de una aplicación lineal y
    $\lambda=\lambda_\#\id_{\simplex{m}}$, por el lema de naturalidad lineal
    \[
      \Sd(\sigma\lambda)=(\sigma\lambda)_\#\Sd\id_{\simplex{m}}
      =\sigma_\#\lambda_\#\Sd\id_{\simplex{m}}
      =\sigma_\#\Sd\lambda_\#\id_{\simplex{m}}
      =\sigma_\#\Sd\lambda,
    \]
    y análogamente para $T$.
  \end{proof}

  Por tanto no hay ambigüedad en la notación $\Sd$ y $T$.
\end{frame}

\subsection{Propiedades}

\begin{frame}
  \frametitle{El operador $\Sd$ es un morfismo de complejos}

  \begin{proposition}
    La aplicación $\Sd\colon\Sing_\bullet(X)\to\Sing_\bullet(X)$ es un morfismo de complejos.
  \end{proposition}

  \begin{proof}
    Sea $\sigma\colon\simplex{n}\to X$ con $n\geq1$ (si $n=0$ ambos miembros
    son nulos). Como $\partial\id_{\simplex{n}}=\sum_i(-1)^i\delta_i$ con $\delta_i\colon\simplex{n-1}\to\simplex{n}$ la inclusión de la $i$-ésima cara, que es
    lineal,
    \begin{align*}
      \partial\Sd\sigma
      &=\partial\sigma_\#\Sd\id_{\simplex{n}}
      =\sigma_\#\partial\Sd\id_{\simplex{n}}
      =\sigma_\#\Sd\partial\id_{\simplex{n}}\\
      &=\sum_{i=0}^n(-1)^i\,\sigma_\#\Sd\delta_i
      =\sum_{i=0}^n(-1)^i\,\Sd(\sigma\delta_i)
      =\sum_{i=0}^n(-1)^i\,\Sd\partial_i\sigma
      =\Sd\partial\sigma.
    \end{align*}
    Hemos usado que $\Sd$ es un morfismo de complejos en $\LC_\bullet(\simplex{n})$ y
    el lema anterior con $\lambda=\delta_i$.
  \end{proof}
\end{frame}

\begin{frame}
  \frametitle{El operador $\Sd$ es homotópico a la identidad}

  \begin{proposition}
    En $\Sing_\bullet(X)$, se tiene que $\partial T+T\partial=\id-\Sd$ .
  \end{proposition}

  \begin{proof}
    Sea $\sigma\colon\simplex{n}\to X$. Por la identidad lineal
    $\partial T\id_{\simplex{n}}=\id_{\simplex{n}}-\Sd\id_{\simplex{n}}-T\partial\id_{\simplex{n}}$ (si $n=0$,
    $T\partial\id_{\simplex{0}}=T[\;]=0$),
    \begin{align*}
      \partial T\sigma
      &=\partial \sigma_\#T\id_{\simplex{n}}
      =\sigma_\#\partial T\id_{\simplex{n}}
      =\sigma_\#(\id_{\simplex{n}}-\Sd\id_{\simplex{n}}-T\partial\id_{\simplex{n}})\\
      &=\sigma-\Sd\sigma-\sum_{i=0}^n(-1)^i\,\sigma_\#T\delta_i
      =\sigma-\Sd\sigma-\sum_{i=0}^n(-1)^i\,T(\sigma\delta_i)\\
      &=\sigma-\Sd\sigma-\sum_{i=0}^n(-1)^i\,T\partial_i\sigma
      =\sigma-\Sd\sigma-T\partial\sigma.
    \end{align*}
  \end{proof}
\end{frame}

\begin{frame}
  \frametitle{Naturalidad}

  \begin{proposition}
    Sea $g\colon X\to X'$ continua. Entonces $g_\#\Sd=\Sd g_\#$ y
    $g_\#T=Tg_\#$.
  \end{proposition}

  \begin{proof}
    Para $\sigma\colon\simplex{n}\to X$,
    \[
      g_\#\Sd\sigma=g_\#\sigma_\#\Sd\id_{\simplex{n}}=(g\sigma)_\#\Sd\id_{\simplex{n}}=\Sd(g\sigma)
      =\Sd g_\#\sigma,
    \]
    y análogamente para $T$.
  \end{proof}

  \medskip

  En particular, si $\sigma$ tiene imagen en $A\subseteq X$, lo mismo ocurre con
  los símplices que aparecen en $\Sd\sigma$ y $T\sigma$. Por tanto $\Sd$ y $T$
  se restringen a $\Sing_\bullet(A)$.
\end{frame}

\begin{frame}
  \frametitle{Resumen}

  Para todo espacio $X$ hemos construido
  \begin{itemize}
    \item un morfismo de complejos $\Sd\colon\Sing_\bullet(X)\to\Sing_\bullet(X)$,
    \item una homotopía de cadenas $T$ con $\partial T+T\partial=\id-\Sd$,
  \end{itemize}
  naturales en $X$, que no sacan las cadenas de ningún subespacio, y tales que
  $\Sd\sigma=\sigma_\#\Sd\id_{\simplex{n}}$ para todo símplice singular
  $\sigma\colon\simplex{n}\to X$.

  \medskip

  Siguiente paso: iterar $\Sd$ hasta que los símplices sean tan pequeños que
  cada uno caiga en algún elemento del recubrimiento.
\end{frame}
